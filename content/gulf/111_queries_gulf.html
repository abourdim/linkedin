Partie 8 — Recherche booleenne avancee

LinkedIn accepte les operateurs AND, OR et les guillemets "..." dans la barre de recherche.

Requete generale

"embedded" AND ("C++" OR "firmware") AND ("UAE" OR "Saudi Arabia" OR "Qatar")

Par technologie

  • ("RTOS" OR "FreeRTOS" OR "VxWorks") AND ("Dubai" OR "Abu Dhabi" OR "Riyadh")
  • ("ARM Cortex" OR "STM32" OR "NXP") AND ("UAE" OR "Saudi Arabia")
  • ("embedded Linux" OR "Yocto" OR "Buildroot") AND ("Middle East" OR "Gulf")

Par secteur

  • "embedded" AND ("oil" OR "gas" OR "energy") AND ("UAE" OR "Saudi Arabia")
  • "embedded" AND ("aerospace" OR "defense" OR "DO-178") AND ("Middle East")
  • "embedded" AND ("EV charging" OR "V2G" OR "ISO 15118") AND ("Gulf" OR "UAE")
  • "embedded" AND ("smart city" OR "NEOM" OR "IoT") AND ("Saudi Arabia" OR "UAE")
